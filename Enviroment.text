# 1. Cài đặt các thư viện cần thiết để build Python

# 2. Tải mã nguồn Python 3.10.11

wget https://www.python.org/ftp/python/3.10.11/Python-3.10.11.tgz
tar -xf Python-3.10.11.tgz
cd Python-3.10.11

# 3. Biên dịch (Lưu ý dùng altinstall để KHÔNG ghi đè Python hệ thống)

./configure --enable-optimizations
make -j4  
sudo make altinstall
