#=================================================================
# 1. Tạo file service hệ thống
#=================================================================

sudo nano /etc/systemd/system/traffic_ai.service
#copy nội dung ở traffic_ai.serveic --> Ctrl+O --> Enter --> Ctrl+X

#=================================================================
# 2. Bắt đầu chạy hệ thống AI ngầm
#=================================================================
sudo systemctl daemon-reload
sudo systemctl enable traffic_ai.service
#Tự thay đổi path dẫn và tên user tùy theo mỗi pi
#=================================================================
# 3. Lệnh chạy nhanh nếu muốn 
#=================================================================
nano ~/.bashrc
#Kéo xuống cuối past vào
===============================================================================================================
alias traffic-on="sudo systemctl start traffic_ai.service && echo '[OK] Đã bật 2.4Ghz & Đang chạy Traffic AI...'"
alias traffic-off="sudo systemctl stop traffic_ai.service && echo '[OK] Đã dừng Traffic AI & Giải phóng tài nguyên'"
alias traffic-reset="sudo systemctl restart traffic_ai.service && echo '[OK] Đã khởi động lại toàn bộ hệ thống Traffic AI'"
alias traffic-status="systemctl status traffic_ai.service"
alias traffic-log="journalctl -u traffic_ai.service -n 50 -f"
=================================================================================
# Lưu và cập nhật lại terminal
source ~/.bashrc
==================================================================================
traffic-on : Bật
traffic-off : Tắt
traffic-reset : Khởi động lại hệ thống
traffic-log : log debug của hệ thống
traffic-status : Trạng thái hệ thống
