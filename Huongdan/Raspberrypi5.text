#================================================================
# SYSTEMD SERVICE SETUP — Traffic AI & OLED
#================================================================

# 0. QUYỀN & I2C (chạy 1 lần)
sudo raspi-config nonint do_i2c 0          # bật I2C
sudo usermod -aG i2c $USER                 # cần reboot để có hiệu lực
sudo chmod +x ~/traffic_ai/*.py            # cấp quyền script
i2cdetect -y 1                             # kiểm tra OLED (thường 0x3C)

# 1. TẠO SERVICE FILES
sudo nano /etc/systemd/system/traffic_ai.service
sudo nano /etc/systemd/system/oled.service
# (Paste nội dung → Ctrl+O → Enter → Ctrl+X)

# 2. KÍCH HOẠT
sudo systemctl daemon-reload
sudo systemctl enable traffic_ai.service oled.service
sudo systemctl start  traffic_ai.service oled.service

# 3. KIỂM TRA
sudo systemctl status traffic_ai.service oled.service
sudo journalctl -u traffic_ai.service -u oled.service -f   # log realtime

# 4. REBOOT TEST
sudo reboot
# Sau reboot: sudo systemctl status traffic_ai.service oled.service

#================================================================
# ALIAS (thêm vào ~/.bashrc)
#================================================================
nano ~/.bashrc   # paste đoạn dưới vào cuối file

alias traffic-on="sudo systemctl start   traffic_ai.service oled.service && echo '[OK] Đã bật'"
alias traffic-off="sudo systemctl stop   traffic_ai.service oled.service && echo '[OK] Đã dừng'"
alias traffic-reset="sudo systemctl restart traffic_ai.service oled.service && echo '[OK] Đã restart'"
alias traffic-status="systemctl status   traffic_ai.service oled.service --no-pager -n 0"
alias traffic-log="journalctl -u traffic_ai.service -u oled.service -n 50 -f"
alias traffic-log-err="journalctl -u traffic_ai.service -u oled.service -p err -n 50 -f"
alias traffic-i2c="i2cdetect -y 1"
alias traffic-fix-perm="sudo chmod +x ~/traffic_ai/*.py && echo '[OK] chmod done'"

source ~/.bashrc   # áp dụng ngay

#================================================================
# QUICK REFERENCE
# traffic-on        bật cả 2 service
# traffic-off       tắt cả 2 service
# traffic-reset     restart cả 2
# traffic-status    xem trạng thái
# traffic-log       xem log realtime (Ctrl+C thoát)
# traffic-log-err   chỉ xem lỗi ERROR/WARNING
# traffic-i2c       kiểm tra OLED trên I2C bus
# traffic-fix-perm  chmod +x lại toàn bộ .py
#================================================================
